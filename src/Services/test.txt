<Select
  value={selectedAgentId !== null ? selectedAgentId.toString() : ""}
  label="Agente"
  onChange={(e) => {
    const value = e.target.value as string;
    const selected = value === "" ? null : parseInt(value, 10);
    setSelectedAgentId?.(selected);
    fetchRatingByAgent?.(selected, 1, 5);
  }}
>
