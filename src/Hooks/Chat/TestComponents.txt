/* eslint-disable @typescript-eslint/no-explicit-any */
import type { MessageDto } from "@/Interfaces/Chat/ChatInterfaces"
import { DownloadMedia, sendMedia, sendText } from "@/Services/MessageService"
import { Check, CheckCheck } from "lucide-react"
import { useCallback, useEffect, useMemo, useRef, useState, type ChangeEvent } from "react"
import useMessages from "@/Hooks/useMessages"
import { useAudioRecorder } from "@/Hooks/Chat/useAudioRecorder"
import { toast } from "react-toastify"
import React from "react"

type ReplyTarget = {
  messageId: number
  externalId: string
  author: string
  preview: string
}


type UseChatWindowArgs = {
  conversationId?: number
  userId?: number
  conversationStatus?: string | null
  selectedParam?: unknown
  highlightTerm?: string
}

export const useChatWindow = (args: UseChatWindowArgs = {},   validFrom?: Date | null, validTo?: Date | null) => {
  const LONG_PRESS_MS = 400

  const [conversationId, setConversationId] = useState<number>(Number(args.conversationId ?? 0))
  const [userId, setUserId] = useState<number>(Number(args.userId ?? 0))
  
  const messages = useMessages(validFrom, validTo, conversationId)

  const [text, setText] = useState("")
  const [selectedOptionState, _setSelectedOption] = useState<string | null>(null)
  const selectedOptionRef = useRef<string | null>(null)

  const [file, setFile] = useState<File | null>(null)
  const [sending, setSending] = useState(false)
  const [ismaximize, setismaximize] = useState(false)
  const [previewImageUrl, setPreviewImageUrl] = useState<string | null>(null)

  const hasMedia = (m: MessageDto) => (m.attachments?.length ?? 0) > 0
  const hasText = (m: MessageDto) => typeof m.content === "string" && m.content.trim().length > 0

  const bottomRef = useRef<HTMLDivElement>(null)
  const fileInputRef = useRef<HTMLInputElement>(null)

  const { isRecording, audioBlob, recordingTime, startRecording, stopRecording, clearRecording } = useAudioRecorder()

  const [quickMenuOpen, setQuickMenuOpen] = useState(false)
  const [quickMenuPos, setQuickMenuPos] = useState<{ x: number; y: number }>({ x: 0, y: 0 })
  const [quickMenuMsg, setQuickMenuMsg] = useState<MessageDto | null>(null)

  const longPressTimerRef = useRef<number | null>(null)
  const popoverRef = useRef<HTMLDivElement | null>(null)
  const textAreaRef = useRef<HTMLTextAreaElement | null>(null)

  const messageRefs = useRef<Map<number, HTMLDivElement>>(new Map())
  const [highlightedId, setHighlightedId] = useState<number | null>(null)

const [searchTerm, setSearchTerm] = useState<string>((args.highlightTerm ?? "").trim());
useEffect(() => { setSearchTerm((args.highlightTerm ?? "").trim()) }, [args.highlightTerm]);

//const escapeRegExp = (s: string) => s.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");

  const registerMessageRef = (id: number) => (el: HTMLDivElement | null) => {
    if (el) messageRefs.current.set(id, el)
    else messageRefs.current.delete(id)
  }

  const scrollToMessage = (id?: number) => {
    if (!id) return
    const el = messageRefs.current.get(id)
    if (!el) return
    el.scrollIntoView({ behavior: "smooth", block: "center" })
    setHighlightedId(id)
    window.setTimeout(() => setHighlightedId(null), 1600)
  }

  useEffect(() => {
    setConversationId(Number(args.conversationId ?? 0))
    setUserId(Number(args.userId ?? 0))
  }, [args.conversationId, args.userId])

  useEffect(() => {
    const onClickOutside = (e: MouseEvent) => {
      if (!popoverRef.current) return
      if (!popoverRef.current.contains(e.target as Node)) {
        setQuickMenuOpen(false)
        setQuickMenuMsg(null)
      }
    }
    if (quickMenuOpen) document.addEventListener("mousedown", onClickOutside)
    return () => document.removeEventListener("mousedown", onClickOutside)
  }, [quickMenuOpen])

  const quickOptions: string[] = useMemo(() => {
    const p = args.selectedParam as any
    if (Array.isArray(p)) return p.filter((v: any) => typeof v === "string")
    if (typeof p === "string") {
      try {
        const parsed = JSON.parse(p)
        if (Array.isArray(parsed)) return parsed.filter((v: any) => typeof v === "string")
      } catch {
        return String(p)
          .split("|")
          .map((s) => s.trim())
          .filter(Boolean)
      }
    }
    if (typeof p?.value === "string") {
      try {
        const parsed = JSON.parse(p.value)
        if (Array.isArray(parsed)) return parsed.filter((v: any) => typeof v === "string")
      } catch {
        return String(p.value)
          .split("|")
          .map((s: string) => s.trim())
          .filter(Boolean)
      }
    }
    if (Array.isArray(p?.value)) return p.value.filter((v: any) => typeof v === "string")
    return []
  }, [args.selectedParam])

  const handleSendAudio = async () => {
    if (!conversationId || sending || !audioBlob) return
    setSending(true)
    try {
      const mime = audioBlob.type || "audio/webm"
      const ext = mime.includes("ogg") ? "ogg" : mime.includes("webm") ? "webm" : mime.includes("mpeg") ? "mp3" : mime.includes("mp4") || mime.includes("aac") ? "m4a" : "bin"
      const audioFile = new File([audioBlob], `grabacion.${ext}`, { type: mime })
      await sendMedia(conversationId, audioFile, "")
      clearRecording()
    } finally {
      setSending(false)
    }
  }

  const visibleMessages = useMemo(
    () => messages.filter((m) => hasMedia(m) || hasText(m) || m.messageType === "Interactive"),
    [messages]
  )

  useEffect(() => {
    bottomRef.current?.scrollIntoView({ block: "end" })
  }, [visibleMessages])

  const handleFileChange = (e: ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0]
    if (!file) return
    let maxSize = 16 * 1024 * 1024
    let errorMsg = "El archivo supera el límite permitido."
    if (file.type.startsWith("application/")) {
      maxSize = 100 * 1024 * 1024
      errorMsg = "Los documentos no pueden superar 100MB."
    } else if (file.type.startsWith("image/") || file.type.startsWith("video/")) {
      maxSize = 16 * 1024 * 1024
      errorMsg = "Las imágenes y videos no pueden superar 16MB."
    } else if (file.type === "image/webp") {
      if (file.size > 100 * 1024 && file.size <= 500 * 1024) {
        maxSize = 500 * 1024
        errorMsg = "Los stickers animados no pueden superar 500KB."
      } else {
        maxSize = 100 * 1024
        errorMsg = "Los stickers estáticos no pueden superar 100KB."
      }
    }
    if (file.size > maxSize) {
      toast.warn(errorMsg)
      e.target.value = ""
      return
    }
    setFile(file)
  }

  const setSelectedOption = (opt: string | null) => {
    selectedOptionRef.current = opt
    _setSelectedOption(opt)
  }

  const [replyTo, setReplyTo] = useState<ReplyTarget | null>(null)

  const beginReply = (m: MessageDto) => {
    const extId = (m as any).externalId || null
    if (!extId) return
    const author = m.senderContactName || m.senderUserName || "Mensaje"
    const preview =
      (typeof m.content === "string" && m.content.trim()) ||
      m.attachments?.[0]?.fileName ||
      (m.attachments?.[0]?.mimeType?.startsWith("image/") ? "[Imagen]" : m.attachments?.[0]?.mimeType?.startsWith("video/") ? "[Video]" : m.attachments?.[0]?.mimeType?.startsWith("audio/") ? "[Audio]" : m.messageType === "Document" ? "[Documento]" : "[Adjunto]")
    setReplyTo({ messageId: m.messageId, externalId: extId, author, preview })
  }

  const cancelReply = () => setReplyTo(null)

  const handlePaste = (e: React.ClipboardEvent<HTMLTextAreaElement>) => {
    if (file) return
    const clipboardItems = Array.from(e.clipboardData.items)
    for (const item of clipboardItems) {
      if (item.kind === "file") {
        const pastedFile = item.getAsFile()
        if (pastedFile) {
          e.preventDefault()
          setFile(pastedFile)
          break
        }
      }
    }
  }

  
    function renderMessageContent(content: string, searchTerm: string) {
      // detecta URLs
      const urlRegex = /(https?:\/\/[^\s]+)/g;

      // divide el texto en partes (texto + URLs)
      const parts = content.split(urlRegex);

      return parts.map((part, i) => {
        if (urlRegex.test(part)) {
          return (
            <a
              key={i}
              href={part}
              target="_blank"
              rel="noopener noreferrer"
              style={{ color: '#1a73e8', textDecoration: 'underline' }}
            >
              {part}
            </a>
          );
        }

        // si usas highlightText para resaltar búsqueda:
        return <React.Fragment key={i}>{highlightText(part, searchTerm)}</React.Fragment>;
      });
    }

  const handleKeyPress = (e: React.KeyboardEvent<HTMLTextAreaElement>) => {
    if (e.key === "Enter" && !e.shiftKey) {
      e.preventDefault()
      handleSend()
    }
  }

  const handleSend = useCallback(async () => {
    if (!conversationId || sending) return
    setSending(true)
    try {
      const msg = text.trim()
      const quick = (selectedOptionRef.current ?? "").trim()
      if (!file && !msg && !quick) return
      const replyExternalId = replyTo?.externalId || null
      const repliedParentMessageId = replyTo?.messageId ?? null
      if (file) {
        await sendMedia(conversationId, file, msg || undefined, replyExternalId, repliedParentMessageId)
        setFile(null)
        setText("")
        if (fileInputRef.current) fileInputRef.current.value = ""
        setReplyTo(null)
        return
      }
      if (quick) {
        await sendText({
          conversationId,
          senderId: userId,
          content: quick,
          messageType: "Text",
          externalId: replyExternalId,
          repliedParentMessageId
        })
        setSelectedOption(null)
        setReplyTo(null)
        return
      }
      if (msg) {
        await sendText({
          conversationId,
          senderId: userId,
          content: msg,
          messageType: "Text",
          externalId: replyExternalId,
          repliedParentMessageId
        })
        setText("")
        setReplyTo(null)
      }
    } finally {
      setSending(false)
    }
  }, [conversationId, file, sending, text, userId, replyTo])

  const handleDownload = async (fileName: string) => {
    try {
      const response = DownloadMedia(fileName)
      const blob = new Blob([(await response).data])
      const url = window.URL.createObjectURL(blob)
      const link = document.createElement("a")
      link.href = url
      link.download = fileName || "archivo"
      document.body.appendChild(link)
      link.click()
      document.body.removeChild(link)
      window.URL.revokeObjectURL(url)
    // eslint-disable-next-line no-empty
    } catch { }
  }

  const getStatusIcon = (m: MessageDto, isOut: boolean) => {
    if (!isOut) return null
    if (m.readAt) return <CheckCheck size={14} className="chat-window__message-status chat-window__message-status--read" />
    if (m.deliveredAt) return <CheckCheck size={14} className="chat-window__message-status chat-window__message-status--delivered" />
    if (m.status === "Sent") return <Check size={14} className="chat-window__message-status chat-window__message-status--sent" />
    return null
  }

  const getMessageClassName = (m: MessageDto) => {
    if (m.senderContactId) return "chat-window__message chat-window__message--incoming"
    if (m.senderUserId && m.senderUserId !== userId) return "chat-window__message chat-window__message--agent"
    return "chat-window__message chat-window__message--outgoing"
  }

  const getMessagePosition = (m: MessageDto) => {
    if (m.senderUserId === userId) return "chat-window__message-group chat-window__message-group--outgoing"
    return "chat-window__message-group chat-window__message-group--incoming"
  }

  const getFileIcon = (mimeType: string) => {
    if (mimeType.startsWith("image/")) return "🖼️"
    if (mimeType.startsWith("video/")) return "🎥"
    if (mimeType.startsWith("audio/")) return "🎵"
    if (mimeType.includes("pdf")) return "📄"
    if (mimeType.includes("word") || mimeType.includes("document")) return "📝"
    if (mimeType.includes("excel") || mimeType.includes("spreadsheet")) return "📊"
    if (mimeType.includes("powerpoint") || mimeType.includes("presentation")) return "📽️"
    return "📎"
  }

  const openImagePreview = (url: string) => {
    setPreviewImageUrl(url)
    setismaximize(true)
  }

  const closeImagePreview = () => {
    setPreviewImageUrl(null)
    setismaximize(false)
  }

  const getCustomoMessageByAgent = async () => {
    toast.info("Cargando respuestas rápidas…")
  }

  const sendQuickNow = async (msg: MessageDto, opt: string) => {
    try {
      await sendText({
        conversationId: Number(conversationId),
        senderId: Number(userId),
        content: opt,
        messageType: "Text",
        externalId: (msg as any)?.externalId ?? null,
        repliedParentMessageId: msg.messageId
      })
    } finally {
      setSelectedOption(null)
      cancelReply()
      setQuickMenuOpen(false)
    }
  }

  const sendQuick = (opt: string) => {
    if (quickMenuMsg) void sendQuickNow(quickMenuMsg, opt)
    else {
      setSelectedOption(opt)
      handleSend()
    }
  }

  const startLongPress = (e: React.PointerEvent, msg: MessageDto) => {
    const rect = (e.target as HTMLElement).getBoundingClientRect()
    const x = (e as any).clientX ?? rect.left + rect.width / 2
    const y = (e as any).clientY ?? rect.top
    if (longPressTimerRef.current) window.clearTimeout(longPressTimerRef.current)
    longPressTimerRef.current = window.setTimeout(() => {
      setQuickMenuPos({ x, y })
      setQuickMenuMsg(msg)
      setQuickMenuOpen(true)
    }, LONG_PRESS_MS) as unknown as number
  }

  const cancelLongPress = () => {
    if (longPressTimerRef.current) {
      window.clearTimeout(longPressTimerRef.current)
      longPressTimerRef.current = null
    }
  }

  const focusTextarea = () => {
    textAreaRef.current?.focus()
  }

  useEffect(() => {
    const onGlobalKeyDown = (e: KeyboardEvent) => {
      if (e.key === "Escape") {
        if (replyTo) {
          e.preventDefault()
          cancelReply()
        } else if (file) {
          e.preventDefault()
          setFile(null)
        }
      }
    }
    window.addEventListener("keydown", onGlobalKeyDown)
    return () => window.removeEventListener("keydown", onGlobalKeyDown)
  }, [replyTo, file])

  const onComposerKeyDown: React.KeyboardEventHandler<HTMLTextAreaElement> = (e) => {
    if (e.key === "Escape") {
      if (replyTo) {
        e.preventDefault()
        cancelReply()
        return
      }
      if (file) {
        e.preventDefault()
        setFile(null)
        return
      }
    }
    if (e.key === "Enter" && !e.shiftKey) {
      e.preventDefault()
      handleSend()
      return
    }
    handleKeyPress(e)
  }

  useEffect(() => {
    const onGlobalEnter = (e: KeyboardEvent) => {
      const el = document.activeElement as HTMLElement | null
      const isTyping = el?.tagName === "TEXTAREA" || el?.tagName === "INPUT" || el?.getAttribute("contenteditable") === "true"
      if (e.key === "Enter" && !e.shiftKey && file && !sending && conversationId && args.conversationStatus !== "Closed") {
        if (!isTyping) {
          e.preventDefault()
          handleSend()
        }
      }
    }
    window.addEventListener("keydown", onGlobalEnter)
    return () => window.removeEventListener("keydown", onGlobalEnter)
  }, [file, sending, conversationId, args.conversationStatus, handleSend])

  const visibleState = {
    quickMenuOpen,
    quickMenuPos,
    quickMenuMsg
  }

  const matchIds = useMemo(() => {
  const term = searchTerm.toLowerCase();
  if (!term) return [];
  return visibleMessages
    .filter(m => hasText(m) && (m.content ?? "").toLowerCase().includes(term))
    .map(m => m.messageId);
}, [visibleMessages, searchTerm]);

const [matchIndex, setMatchIndex] = useState(0);

useEffect(() => {
  if (matchIds.length) {
    setMatchIndex(0);
    scrollToMessage(matchIds[0]);       // auto-scroll a la primera coincidencia
  }
}, [matchIds]);

const gotoMatch = (dir: 1 | -1) => {
  if (!matchIds.length) return;
  const next = (matchIndex + dir + matchIds.length) % matchIds.length;
  setMatchIndex(next);
  scrollToMessage(matchIds[next]);
  setHighlightedId(matchIds[next]);
};

  const highlightText = (text: string, term: string) => {
  const t = term.trim()
  if (!t) return text
  const re = new RegExp(`(${t.replace(/[.*+?^${}()|[\]\\]/g, "\\$&")})`, "ig")
    return text.split(re).map((part, i) =>
      i % 2 === 1
        ? <mark key={i} className="chat-window__hl">{part}</mark>
        : <React.Fragment key={i}>{part}</React.Fragment>
    )
  }

  return {
    setConversationId,
    setUserId,
    messages,
    text,
    setText,
    file,
    setFile,
    sending,
    setSending,
    handleSend,
    bottomRef,
    fileInputRef,
    handleFileChange,
    handlePaste,
    handleKeyPress,
    handleDownload,
    getStatusIcon,
    getMessageClassName,
    getMessagePosition,
    getFileIcon,
    ismaximize,
    setismaximize,
    previewImageUrl,
    openImagePreview,
    closeImagePreview,
    isRecording,
    audioBlob,
    recordingTime,
    startRecording,
    stopRecording,
    clearRecording,
    handleSendAudio,
    visibleMessages,
    hasMedia,
    hasText,
    setSelectedOption,
    selectedOption: selectedOptionState,
    beginReply,
    cancelReply,
    replyTo,
    getCustomoMessageByAgent,
    quickOptions,
    sendQuick,
    startLongPress,
    cancelLongPress,
    focusTextarea,
    onComposerKeyDown,
    popoverRef,
    textAreaRef,
    registerMessageRef,
    scrollToMessage,
    highlightedId,
    visibleState,

    //Buscar texto
    searchTerm,
    setSearchTerm,
    matchIds,
    matchIndex,
    gotoMatch,
    highlightText,
    renderMessageContent
  }
}

import { useState, useEffect, useCallback } from 'react'
import type {
  CustomMessageRequestDto,
  CustomMessageResponseDto,
} from '@/Interfaces/Setting/CustomMessageInterface'
import { CustomMessageServices } from '@/Services/Config/CustomMessageServices'

export function useCustomMessage(autoLoad = true) {
  const [customMessages, setCustomMessages] = useState<CustomMessageResponseDto[]>([])
  const [selectedCustomMessage, setSelectedCustomMessage] = useState<CustomMessageResponseDto | null>(null)
  const [loadingList, setLoadingList] = useState(false)
  const [loadingItem, setLoadingItem] = useState(false)
  const [error, setError] = useState<Error | null>(null)
  const [modalMode, setModalMode] = useState<'create' | 'edit' | 'view'>('create')
  const [isModalOpen, setModalOpen] = useState(false)
  const [loaded, setLoaded] = useState(false)

  const fetchPage = useCallback(async () => {
    setLoadingList(true)
    setError(null)
    try {
      const resp = await CustomMessageServices.getAllAsync()
      setCustomMessages(resp)
      setLoaded(true)
    } catch (err: any) {
      setError(err)
    } finally {
      setLoadingList(false)
    }
  }, [])

  const fetchById = useCallback(async (id: number) => {
    setLoadingItem(true)
    setError(null)
    try {
      const item = await CustomMessageServices.getByIdAsync(id)
      setSelectedCustomMessage(item)
    } catch (err: any) {
      setError(err)
    } finally {
      setLoadingItem(false)
    }
  }, [])

  const create = useCallback(
    async (values: CustomMessageRequestDto) => {
      setLoadingItem(true)
      setError(null)
      try {
        await CustomMessageServices.createAsync(values)
        await fetchPage()
      } catch (err: any) {
        setError(err)
      } finally {
        setLoadingItem(false)
      }
    },
    [fetchPage]
  )

  const update = useCallback(
    async (id: number, values: CustomMessageRequestDto) => {
      setLoadingItem(true)
      setError(null)
      try {
        await CustomMessageServices.updateAsync(id, values)
        await fetchPage()
      } catch (err: any) {
        setError(err)
      } finally {
        setLoadingItem(false)
      }
    },
    [fetchPage]
  )

  const toggleStatus = useCallback(
    async (id: number) => {
      setLoadingItem(true)
      setError(null)
      try {
        const toggled = await CustomMessageServices.toggleStatusAsync(id)
        setCustomMessages(prev =>
          prev.map(item => (item.id === toggled.id ? toggled : item))
        )
      } catch (err: any) {
        setError(err)
      } finally {
        setLoadingItem(false)
      }
    },
    []
  )

  const openCreate = useCallback(() => {
    setSelectedCustomMessage(null)
    setModalMode('create')
    setModalOpen(true)
  }, [])

  const openEdit = useCallback(
    (id: number) => {
      const cache = customMessages.find(x => x.id === id) || null
      setSelectedCustomMessage(cache)
      setModalMode('edit')
      setModalOpen(true)
      fetchById(id)
    },
    [customMessages, fetchById]
  )

  const openView = useCallback(
    (id: number) => {
      const cache = customMessages.find(x => x.id === id) || null
      setSelectedCustomMessage(cache)
      setModalMode('view')
      setModalOpen(true)
      fetchById(id)
    },
    [customMessages, fetchById]
  )

  const closeModal = useCallback(() => {
    setModalOpen(false)
    setSelectedCustomMessage(null)
  }, [])

  const handleSubmit = useCallback(
    (values: CustomMessageRequestDto) => {
      if (modalMode === 'create') create(values)
      else if (modalMode === 'edit' && selectedCustomMessage) update(selectedCustomMessage.id, values)
      closeModal()
    },
    [modalMode, selectedCustomMessage, create, update, closeModal]
  )

  useEffect(() => {
    if (autoLoad) {
      fetchPage()
    }
  }, [autoLoad, fetchPage])

  return {
    customMessages,
    selectedCustomMessage,
    loading: loadingList || loadingItem,
    error,
    modalMode,
    isModalOpen,
    openCreate,
    openEdit,
    openView,
    closeModal,
    handleSubmit,
    toggleSystemParam: toggleStatus,
    create,
    update,
    load: fetchPage,
    loaded,
  }
}

  if (!iso) return "";
  const match = iso.match(/T(\d{2}:\d{2})/);
  return match ? match[1] : "";
};

interface Props {
  conversationId?: number;
  conversationStatus?: string | null;
  userId: number;
  selectedParam: SystemParamResponseDto | null;
  highlightTerm?: string 
  validFrom: Date | null
  validTo: Date | null
}


export const ChatWindow: React.FC<Props> = ({
  conversationId,
  userId,
  conversationStatus,
  selectedParam,
  highlightTerm,
  validFrom,
  validTo
}) => {

  const smsBtnRef = useRef<HTMLDivElement | null>(null);
  const [flyoutOpen, setFlyoutOpen] = useState(false);
  const [showScrollButton, setShowScrollButton] = useState(false);
  const dateRefs = useRef<Record<string, HTMLDivElement | null>>({});
  const [stickyDate, setStickyDate] = useState<string | null>(null);

  const messagesContainerRef = useRef<HTMLDivElement|null>(null);

const formatDuration = (ms: number): string => {
  const totalMinutes = Math.floor(ms / 60000)
  const hours = Math.floor(totalMinutes / 60)
  const minutes = totalMinutes % 60
  return hours > 0 ? `${hours}h ${minutes}m` : `${minutes}m`
}

import React, { useState, useEffect, useMemo, useCallback } from "react";
import { jwtDecode } from "jwt-decode";
import { useDispatch } from "react-redux";
import {
  getConversation,
  closeConversation,
  SetHaltedConversation
} from "@/Services/ConversationService";
import type { ConversationDto } from "@/Interfaces/Chat/ChatInterfaces";
import { openAssignModal } from "@/Context/Slices/assignModalSlice";
import { useSystemParams } from "@/Hooks/Setting/useSystemParams";
interface AuthStorage {
  accessToken: string;
  refreshToken: string;
  expiresAt: string;
  userId: number;
}
interface JwtPayload {
  role: string;
}
type FilterOption = { label: string; value: string; content: string };

const FILTER_ORDER_STORAGE_KEY = "filterOrder";

export function useSupportPage() {
  const dispatch = useDispatch();
  const [convId, setConvId] = useState<number | null>(null);
  const [conversation, setConversation] = useState<ConversationDto | null>(
    null
  );
  const [isToggling, setIsToggling] = useState(false);

  const [filter, setFilter] = useState<string>(
    () => localStorage.getItem("selectedFilter") || "all"
  );

  const [selectedFilter, setSelectedFilter] = useState<string>(
    () => localStorage.getItem("selectedFilter") || "all"
  );
  const [showHistoryModal, setShowHistoryModal] = useState(false);
  const [contactDetailToggle, setContactDetailToggle] = useState(false);
  const [orderModalVisible, setOrderModalVisible] = useState(false);
  const { selectedParam, fetchByName } = useSystemParams();

  const authRaw = localStorage.getItem("auth") || "{}";
  const { accessToken, userId } = JSON.parse(authRaw) as AuthStorage;
  const { role } = jwtDecode<JwtPayload>(accessToken);
  const isAdmin = role.toLowerCase() === "admin";
  let DEFAULT_OPTIONS: FilterOption[];

  const [detailToggle, setDetailToggle] = useState<boolean>(false);

  if (isAdmin) {
    DEFAULT_OPTIONS = [
      { label: "Todos", value: "all", content: "Todos" },
      { label: "Bot", value: "bot", content: "Bot" },
      { label: "Pendiente", value: "waiting", content: "Pendiente" },
      { label: "Humano", value: "human", content: "Asignado" },
      { label: "Cerrado", value: "closed", content: "Cerrado" },
      { label: "Incompleto", value: "incomplete", content: "Incompleto" },
    ];
  } else {
    DEFAULT_OPTIONS = [
      { label: "Todos", value: "all", content: "Todos" },
      { label: "Humano", value: "human", content: "Humano" },
      { label: "Cerrado", value: "closed", content: "Cerrado" },
      { label: "Incompleto", value: "incomplete", content: "Incompleto" },
    ];
  }

  // Cargar orden de filtros desde localStorage o usar el predeterminado
  const [filterOrder, setFilterOrder] = useState<string[]>(() => {
    const stored = localStorage.getItem(FILTER_ORDER_STORAGE_KEY);
    if (stored) {
      try {
        return JSON.parse(stored);
      } catch {
        /* ignore */
      }
    }
    return DEFAULT_OPTIONS.map((o) => o.value);
  });

  useEffect(() => {
    setFilter(selectedFilter);
  }, [selectedFilter]);

  const sortedOptions = useMemo(() => {
    return filterOrder
      .map((val) => DEFAULT_OPTIONS.find((o) => o.value === val))
      .filter((o): o is FilterOption => !!o);
  }, [DEFAULT_OPTIONS, filterOrder]);

  useEffect(() => {

    if (!convId) {
      setConversation(null);
      return;
    }
    getConversation(convId)
      .then((res) => setConversation(res.data.data))
      .catch(console.error);

      fetchByName('OpcionesRapidas')
  }, [convId]);
  

  const handleAfterAssign = () => {
    if (!convId) return;
    getConversation(convId)
      .then((res) => setConversation(res.data.data))
      .catch(console.error);
  };

  const handleAssignClick = () => {
    if (!convId || !conversation) return;
    if (isAdmin) {
      dispatch(openAssignModal(conversation));
    } else {
      closeConversation(convId).then(handleAfterAssign).catch(console.error);
    }
  };

  
  const handleSetHaltedConversation = useCallback(async () => {
    if (!convId || !conversation || !isAdmin) return
    try {
      await SetHaltedConversation(convId)
      setConversation(prev =>
        prev ? { ...prev, isHalted: !prev.isHalted } : prev
      )
    } catch (e) {
      console.error(e)
    }
  }, [convId, conversation, isAdmin])

  const handleOptionChange = (e: React.ChangeEvent<HTMLInputElement>) => {
    const val = e.target.value;
    setFilter(val);
    setSelectedFilter(val);
    localStorage.setItem("selectedFilter", val);
  };

  //Funcion separar los primeros 3 numeros 505 del resto del numero.
  function extraerNumero(cadena: string): string {

    return cadena.slice(0, 3) + "-" + cadena.slice(3);
  }
  const saveOrder = (newOrder: string[]) => {
    setFilterOrder(newOrder);
    localStorage.setItem(FILTER_ORDER_STORAGE_KEY, JSON.stringify(newOrder));
  };

  return {
    conversation,
    setOrderModalVisible,
    saveOrder,
    filterOrder,
    DEFAULT_OPTIONS,
    orderModalVisible,
    setShowHistoryModal,
    showHistoryModal,
    setContactDetailToggle,
    contactDetailToggle,
    detailToggle,
    setDetailToggle, 
    convId,
    isAdmin,
    handleAssignClick,
    sortedOptions,
    handleOptionChange,
    setConvId,
    filter,
    userId,
    extraerNumero,
    selectedParam,
    isToggling,
    setIsToggling,
    handleSetHaltedConversation
  };
}
